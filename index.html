function buildHistoryHTML(mainCsv, subCsvList, folderName) { let html = `
${mainCsv.code} ${folderName}`; if (mainCsv.operators.length > 0) { html += ` 業者：${mainCsv.operators.join('、')}`; } html += `
`; const renderTable = (title, histArr) => { if (!histArr || histArr.length === 0) return ''; let t = title ? `
▸ ${title}
` : ''; t += ``; histArr.forEach(h => { const yrStr = h.startYear === h.endYear ? `${h.startYear}` : `${h.startYear}–${h.endYear}`; t += ``; }); t += `
年份 業者 線名 區間 里程
${yrStr} ${h.op || '—'} ${h.rn || '—'} ${h.sec || '—'} ${h.mi || '—'}
`; return t; }; if (mainCsv.history.length > 0) { html += renderTable('', mainCsv.history); } else { html += `
（年報無統計資料之專用線／待考路線）
`; } subCsvList.forEach(sub => { if (sub.history.length > 0) { html += renderTable(`相關路線：${sub.code} ${sub.refName}`, sub.history); } }); return html; }
